3 Introduction

This book explores the writing style of Charles Dickens using three of his public-domain novels:

  • A Tale of Two Cities
  • Great Expectations
  • Oliver Twist

The goals of the project are:

  1. Download and prepare clean text data using R.
  2. Perform descriptive text analysis (frequencies, chapters, patterns).
  3. Use linear regression to test whether Dickens shows consistent chapter structure, such as:
    • Words per chapter
    • Sentences per chapter
    • Words per sentence
  4. Visualize Dickens’s writing patterns using multiple plots.

By the end, we will show whether Dickens has a consistent writing “signature” across books.

# --- Download Dickens Books ---
dickens_books <- gutenberg_download(
  c(98, 1400, 730),
  meta_fields = c("title", "author"),
  strip = TRUE
)

# --- Detect Chapters ---
dickens_books <- dickens_books %>%
  mutate(chapter = cumsum(str_detect(
    text, regex("^chapter\\s+[xivlcdm0-9]+", ignore_case = TRUE)
  ))) %>%
  filter(chapter > 0)

# --- Compute Chapter Stats ---
chapter_stats <- dickens_books %>%
  group_by(title, chapter) %>%
  summarise(
    words = sum(str_count(text, boundary("word"))),
    sentences = sum(str_count(text, boundary("sentence"))),
    .groups = "drop"
  ) %>%
  mutate(words_per_sentence = ifelse(sentences > 0, words / sentences, NA))

# --- Tokenize & Remove Stopwords ---
tidy_dickens <- dickens_books %>%
  unnest_tokens(word, text) %>%
  anti_join(stop_words, by = "word") %>%
  mutate(word = wordStem(word))

# 5. BIGRAM NETWORK (FIXED FOR NA)
# ==========================================

bigrams <- dickens_books %>%
  unnest_tokens(bigram, text, token = "ngrams", n = 2) %>%
  separate(bigram, into = c("word1", "word2"), sep = " ", fill = "right") %>%
  filter(
    !is.na(word1),
    !is.na(word2),
    word1 != "",
    word2 != "",
    !word1 %in% stop_words$word,
    !word2 %in% stop_words$word
  )

bigram_counts <- bigrams %>%
  count(word1, word2, sort = TRUE) %>%
  filter(n > 25)

# If still empty (rare), tighten filter:
if(nrow(bigram_counts) == 0){
  bigram_counts <- bigrams %>% count(word1, word2, sort=TRUE) %>% filter(n > 10)
}

graph <- bigram_counts %>%
  as_tbl_graph()

ggraph(graph, layout = "fr") +
  geom_edge_link(aes(width = n), alpha = 0.3) +
  geom_node_point(color = "steelblue", size = 3) +
  geom_node_text(aes(label = name), repel = TRUE) +
  theme_void() +
  labs(title = "Bigram Network (Frequent 2-Word Pairs)")
